<script>
    function togglePass() {
        const passField = document.getElementById('passInput');
        const toggleBtn = document.querySelector('.toggle-btn');
        if (passField.type === "password") {
            passField.type = "text";
            toggleBtn.innerText = "🙈";
        } else {
            passField.type = "password";
            toggleBtn.innerText = "👁️";
        }
    }

    function checkPassword() {
        const pass = document.getElementById('passInput').value;
        const correctPass = "CHTFILECONVERTER"; 
        if (pass === correctPass) {
            document.getElementById('passwordOverlay').style.display = 'none';
            document.getElementById('mainUI').style.display = 'block';
        } else {
            alert("Wrong Password! Get it from Telegram.");
        }
    }

    const mainMenu = document.getElementById('mainMenu');
    const subSection = document.getElementById('subSection');
    const subSelect = document.getElementById('subCategory');
    const mainSelect = document.getElementById('mainCategory');
    const status = document.getElementById('status');

    const optionsData = {
        'f2i': [{val: 'jpeg', text: 'PNG to Image'}, {val: 'png', text: 'JPG to Image'}, {val: 'webp', text: 'WebP to Image'}],
        'i2f': [{val: 'jpeg', text: 'Image to JPG'}, {val: 'png', text: 'Image to PNG'}, {val: 'webp', text: 'Image to WebP'}],
        'v2a': [{val: 'mp3', text: 'MP4 to MP3'}],
        'f2f': [{val: 'jpeg', text: 'PNG to JPG'}, {val: 'png', text: 'JPG to PNG'}, {val: 'webp', text: 'JPG to WebP'}, {val: 'png', text: 'WebP to PNG'}]
    };

    function openSubMenu() {
        const selected = mainSelect.value;
        if (!selected) return;
        subSelect.innerHTML = '';
        optionsData[selected].forEach(opt => {
            const el = document.createElement('option');
            el.value = opt.val; el.textContent = opt.text;
            subSelect.appendChild(el);
        });
        mainMenu.style.display = 'none';
        subSection.style.display = 'block';
    }

    function goBack() {
        mainMenu.style.display = 'block';
        subSection.style.display = 'none';
        mainSelect.value = "";
    }

    const { createFFmpeg, fetchFile } = FFmpeg;
    const ffmpeg = createFFmpeg({ log: false });
    const convertBtn = document.getElementById('convertBtn');
    const fileInput = document.getElementById('fileInput');

    convertBtn.onclick = async () => {
        const file = fileInput.files[0];
        const format = subSelect.value;
        if (!file) { status.innerText = "⚠️ Upload file!"; return; }
        status.innerText = "Processing... ⏳";
        convertBtn.disabled = true;

        try {
            if (format === 'mp3') {
                if (!ffmpeg.isLoaded()) await ffmpeg.load();
                ffmpeg.FS('writeFile', 'input', await fetchFile(file));
                await ffmpeg.run('-i', 'input', 'output.mp3');
                const data = ffmpeg.FS('readFile', 'output.mp3');
                download(new Blob([data.buffer], { type: 'audio/mp3' }), "CHT_Converted.mp3");
            } else {
                const reader = new FileReader();
                reader.onload = (e) => {
                    const img = new Image();
                    img.onload = () => {
                        const canvas = document.getElementById('tempCanvas');
                        canvas.width = img.width; canvas.height = img.height;
                        canvas.getContext('2d').drawImage(img, 0, 0);
                        download(canvas.toDataURL(`image/${format}`), `CHT_file.${format}`);
                    };
                    img.src = e.target.result;
                };
                reader.readAsDataURL(file);
            }
            status.innerText = "✅ Done!";
            setTimeout(() => { status.innerText = ""; }, 3000);
        } catch (err) { status.innerText = "❌ Error!"; }
        convertBtn.disabled = false;
    };

    function download(url, name) {
        const a = document.createElement('a');
        a.href = typeof url === 'string' ? url : URL.createObjectURL(url);
        a.download = name; a.click();
    }
